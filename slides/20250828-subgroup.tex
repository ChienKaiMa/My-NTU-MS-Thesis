\begin{frame}
    \frametitle{Trade-off MED and UQSD}
    \begin{itemize}
        \item Here, $P$ is the probability matrix for the POVM defined by the set of
              operators $\{\Pi_i\}$, and $Q$ is the target UQSD probability matrix.
        \item $\|P-Q\|_{1}$ is the $L_{1}$ distance between $P$ and $Q$.
        \item $1/w \leq a$ is OptUQSD, where $a$ is usually 2.
        \item $1/w \geq b$ is MED, where the value $b$ varies case by case.
    \end{itemize}
    \begin{align}
        \label{eq:roland_obj}
         & \underset{\{\Pi\}}{\text{maximize}} &  & P_{\text{succ}} - w \|P-Q\|_{1}             \\
         & \text{subject to}                   &  & \Pi_{i} \succeq 0, \quad i = 1, \ldots, k+1 \\
         &                                     &  & \sum_{i=1}^{k+1} \Pi_i = I
    \end{align}
\end{frame}
